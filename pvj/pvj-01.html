spotify_clean <- spotify %>%

    hour = as.integer(format(datetime, "%H")),
    day = format(datetime, "%A"),
    month = format(datetime, "%Y-%m"),
    minutesPlayed = msPlayed / 60000
  )

Code
library(ggplot2)
library(tidyr)
library(stringr)

monthly_top <- monthly_artist %>%
  filter(artistName %in% top_artists) %>%
  complete(
    month,
    artistName,
    fill = list(minutes = 0)
  )

# Order months chronologically by actual date, not by month-number-then-year
month_order <- monthly_top %>%
  distinct(month) %>%
  mutate(month_date = as.Date(paste0(month, "-01"))) %>%
  arrange(month_date) %>%
  pull(month)

# Two-line x-axis labels: month on top, year only where it changes
month_labels <- sapply(seq_along(month_order), function(i) {
  d   <- as.Date(paste0(month_order[i], "-01"))
  mon <- format(d, "%b")
  yr  <- format(d, "%Y")
  if (i == 1 || format(as.Date(paste0(month_order[i - 1], "-01")), "%Y") != yr) {
    paste0(mon, "\n", yr)
  } else {
    mon
  }
})
names(month_labels) <- month_order

monthly_top <- monthly_top %>%
  mutate(
    month = factor(month, levels = month_order),
    artistName = factor(artistName, levels = rev(top_artists)),
    minutes_bin = cut(
      minutes,
      breaks = c(-1, 25, 75, 150, 250, Inf),
      labels = c("0–25", "25–75", "75–150", "150–250", "250+")

  mon <- format(d, "%b")
  yr  <- format(d, "%Y")
  if (i == 1 || format(as.Date(paste0(month_order[i - 1], "-01")), "%Y") != yr) {
    paste0(mon, "\n", yr)
  } else {
    mon
  }
})
